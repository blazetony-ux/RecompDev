cmake_minimum_required(VERSION 3.16)
project(RecompDevHostDemo LANGUAGES CXX)
add_executable(RecompDevHostDemo StandaloneHost.cpp)
set_property(TARGET RecompDevHostDemo PROPERTY CXX_STANDARD 17)
if(WIN32)
 target_link_libraries(RecompDevHostDemo PRIVATE user32 gdi32 winmm)
endif()
add_executable(RecompDevUniversalHost UniversalHost.cpp)
target_compile_features(RecompDevUniversalHost PRIVATE cxx_std_17)
if(WIN32)
 target_link_libraries(RecompDevUniversalHost PRIVATE user32 gdi32 winmm)
endif()
