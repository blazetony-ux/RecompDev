cmake_minimum_required(VERSION 3.17)
project(RecompDevNativeGame LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT RECOMPDEV_SOURCE_ROOT)
 message(FATAL_ERROR "RECOMPDEV_SOURCE_ROOT required for native plugins")
endif()
set(R "${RECOMPDEV_SOURCE_ROOT}")
add_library(rd_video_sms_mode4 SHARED "${R}/video_sms/src/mode4.cpp")
target_compile_definitions(rd_video_sms_mode4 PRIVATE RD_SMS_MODE4_BUILD)
target_include_directories(rd_video_sms_mode4 PUBLIC "${R}/video_sms/include")
add_library(rd_sms_native_runtime SHARED "${R}/runtime_sms/src/runtime.cpp")
target_compile_definitions(rd_sms_native_runtime PRIVATE RD_SMS_NATIVE_BUILD)
target_include_directories(rd_sms_native_runtime PUBLIC "${R}/runtime_sms/include")
target_link_libraries(rd_sms_native_runtime PRIVATE rd_video_sms_mode4)
add_library(rd_runtime_host SHARED "${R}/runtime_host/src/host.cpp")
target_compile_definitions(rd_runtime_host PRIVATE RD_RUNTIME_HOST_BUILD)
target_include_directories(rd_runtime_host PUBLIC "${R}/runtime_host/include")
target_include_directories(rd_runtime_host PRIVATE "${R}/runtime_sms/include")
if(WIN32)
 target_link_libraries(rd_runtime_host PRIVATE user32 gdi32 winmm)
endif()
add_executable(RecompDevNativePilot main.cpp translated.cpp)
add_executable(RecompDevNativeApp interactive.cpp translated.cpp)
target_include_directories(RecompDevNativePilot PRIVATE "${R}/runtime_sms/include" "${R}/runtime_sms/templates" "${R}/video_sms/include")
target_include_directories(RecompDevNativeApp PRIVATE "${R}/runtime_sms/include" "${R}/runtime_sms/templates" "${R}/video_sms/include" "${R}/runtime_host/include")
target_link_libraries(RecompDevNativePilot PRIVATE rd_sms_native_runtime rd_video_sms_mode4)
target_link_libraries(RecompDevNativeApp PRIVATE rd_sms_native_runtime rd_video_sms_mode4 rd_runtime_host)
if(WIN32 AND MSVC)
 target_link_options(RecompDevNativePilot PRIVATE /STACK:8388608)
 target_link_options(RecompDevNativeApp PRIVATE /STACK:8388608)
endif()
